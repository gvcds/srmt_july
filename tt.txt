<configuration>
  <system.webServer>
    
    <!-- Configura o IISNode para o server.js -->
    <handlers>
      <add name="iisnode" path="server.js" verb="*" modules="iisnode" />
    </handlers>
    
    <rewrite>
      <rules>
        <!-- Redireciona TUDO para o server.js gerenciar -->
        <rule name="NextJS">
          <match url="/*" />
          <action type="Rewrite" url="server.js" />
        </rule>
      </rules>
    </rewrite>

    <!-- Força o Node a rodar em modo de produção -->
    <iisnode node_env="production" />

    <!-- Deixa o Next.js cuidar das páginas de erro (ex: 404) -->
    <httpErrors existingResponse="PassThrough" />
    
  </system.webServer>
</configuration>
